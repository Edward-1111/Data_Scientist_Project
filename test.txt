def parse_number(value):
    if pd.isna(value):
        return np.nan

    value = str(value).strip().lower()

    if value in ["", "na", "n/a", "unknown", "?", "-", "null"]:
        return np.nan

    value = value.replace(",", "")
    value = value.replace("$", "")
    value = value.replace("usd", "")

    if value.endswith("k"):
        return float(value[:-1]) * 1000

    return float(value)
